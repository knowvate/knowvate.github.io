# Chris Coats Portfolio - Robots Configuration

# Allow all crawlers to index public pages
User-agent: *
Allow: /
Allow: /projects
Allow: /experience
Allow: /contact

# Disallow authenticated/protected content
Disallow: /extra
Disallow: /admin
Disallow: /login
Disallow: /auth/

# Articles: published articles are public and indexable; only the
# auth-gated editor routes are excluded. Draft articles are additionally
# excluded per-page via a <meta name="robots" content="noindex"> tag
# (the API itself already never serves draft content to anonymous
# requests, so this is defense in depth, not the only protection).
Disallow: /articles/new
Disallow: /articles/*/edit

# Sitemap location (update this URL when deployed)
Sitemap: https://knowvate.github.io/sitemap.xml
